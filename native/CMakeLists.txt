cmake_minimum_required(VERSION 3.25)
project(gTR LANGUAGES CXX RC)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Widgets Multimedia MultimediaWidgets Network)

set(release_version_file "${CMAKE_CURRENT_SOURCE_DIR}/../gtr/__init__.py")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${release_version_file}")
file(STRINGS "${release_version_file}" release_number REGEX "^BETA_NUMBER = [0-9]+$")
file(STRINGS "${release_version_file}" release_fix REGEX "^BETA_FIX = [0-9]+$")
if(NOT release_number OR NOT release_fix)
    message(FATAL_ERROR "gTR release version is missing")
endif()
string(REGEX REPLACE "^BETA_NUMBER = " "" GTR_BETA_NUMBER "${release_number}")
string(REGEX REPLACE "^BETA_FIX = " "" GTR_FIX_NUMBER "${release_fix}")
set(GTR_FIX_LABEL "")
if(GTR_FIX_NUMBER GREATER 1)
    set(GTR_FIX_LABEL " (Fix ${GTR_FIX_NUMBER})")
elseif(GTR_FIX_NUMBER EQUAL 1)
    set(GTR_FIX_LABEL " (Fix)")
endif()
set(GTR_ICON "${CMAKE_CURRENT_SOURCE_DIR}/../gtr/assets/gtr.ico")
configure_file(release.hpp.in release.hpp @ONLY)
configure_file(windows.rc.in windows.rc @ONLY)

add_executable(gTR-shell WIN32 main.cpp window.cpp owner_window.cpp service_client.cpp "${CMAKE_CURRENT_BINARY_DIR}/windows.rc")
target_link_libraries(gTR-shell PRIVATE Qt6::Widgets Qt6::MultimediaWidgets Qt6::Network shell32)
target_include_directories(gTR-shell PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_executable(gTR WIN32 launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/windows.rc")
set_property(TARGET gTR PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
    foreach(target gTR gTR-shell)
        target_compile_options(${target} PRIVATE /utf-8 /W4 /sdl /guard:cf)
        target_link_options(${target} PRIVATE /guard:cf /Brepro /INCREMENTAL:NO)
    endforeach()
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    set(GTR_TEST_PYTHON "${CMAKE_CURRENT_SOURCE_DIR}/../.venv/Scripts/python.exe" CACHE FILEPATH
        "Python interpreter with the gTR dependencies")
    add_executable(protocol-test protocol_test.cpp service_client.cpp)
    target_link_libraries(protocol-test PRIVATE Qt6::Core)
    target_include_directories(protocol-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    if(MSVC)
        target_compile_options(protocol-test PRIVATE /utf-8 /W4 /sdl)
    endif()
    add_test(NAME native-protocol COMMAND protocol-test)
    set_tests_properties(native-protocol PROPERTIES ENVIRONMENT_MODIFICATION
        "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
    if(EXISTS "${GTR_TEST_PYTHON}")
        add_executable(native-integration-test ui_test.cpp window.cpp owner_window.cpp service_client.cpp)
        target_link_libraries(native-integration-test PRIVATE Qt6::Test Qt6::Widgets Qt6::MultimediaWidgets)
        target_include_directories(native-integration-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        if(MSVC)
            target_compile_options(native-integration-test PRIVATE /utf-8 /W4 /sdl)
        endif()
        foreach(size 1366x768 1920x1080 2560x1440)
            foreach(scale 1 2)
                set(name "native-ui-${size}-${scale}")
                add_test(NAME "${name}" COMMAND native-integration-test
                    "${CMAKE_CURRENT_SOURCE_DIR}/.." "${GTR_TEST_PYTHON}" "${size}" "${scale}")
                set_tests_properties("${name}" PROPERTIES
                    TIMEOUT 40
                    ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=${scale};QT_MEDIA_BACKEND=windows"
                    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
            endforeach()
        endforeach()
    endif()
endif()
